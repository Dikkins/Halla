cmake_minimum_required(VERSION 3.22)
project(HallaRussianAddon LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32)
  message(FATAL_ERROR "Halla RU add-on supports Windows x64 only")
endif()
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/upstream/sdk/halla_plugin_api.h")
  message(FATAL_ERROR "Missing v1.1.37 Halla SDK in upstream/")
endif()
find_package(Qt6 6.8.3 EXACT REQUIRED COMPONENTS Core)
add_library(halla_ru SHARED src/halla_ru.cpp)
target_include_directories(halla_ru PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/upstream/sdk")
target_link_libraries(halla_ru PRIVATE Qt6::Core)
set_target_properties(halla_ru PROPERTIES OUTPUT_NAME "halla_ru" PREFIX "")
if(MSVC)
  target_compile_options(halla_ru PRIVATE /W4 /utf-8)
endif()
